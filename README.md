💻 Kurulum Komutları :

sudo apt update
sudo apt install python3-tk

# Proje klasöründe olduğunuzdan emin olun ve venv oluşturun

python3 -m venv venv

# Sanal ortamı aktifleştirin

source venv/bin/activate

pip install --upgrade pip

pip install -r requirements.txt

python3 blueeyes.py
